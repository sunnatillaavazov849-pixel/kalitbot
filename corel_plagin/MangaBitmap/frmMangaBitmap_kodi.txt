Private Sub LoadSettings()
    Select Case GetSetting(REG_APP, REG_SEC, "Scope", "page")
        Case "sel": optSel.Value = True
        Case "all": optAllPages.Value = True
        Case Else: optPage.Value = True
    End Select
    If GetSetting(REG_APP, REG_SEC, "Obj", "bmp") = "all" Then optAllObj.Value = True Else optOnlyBmp.Value = True
    If GetSetting(REG_APP, REG_SEC, "How", "each") = "one" Then optOne.Value = True Else optEach.Value = True
    Select Case GetSetting(REG_APP, REG_SEC, "Mode", "cmyk")
        Case "rgb": optRGB.Value = True
        Case "gray": optGray.Value = True
        Case Else: optCMYK.Value = True
    End Select
    cboDpi.Text = GetSetting(REG_APP, REG_SEC, "Dpi", "300")
    chkAA.Value = (GetSetting(REG_APP, REG_SEC, "AA", "1") = "1")
    chkTransp.Value = (GetSetting(REG_APP, REG_SEC, "Transp", "0") = "1")
    scrBright.Value = 0
    lblBright.Caption = "0"
End Sub

Private Sub SaveSettings()
    On Error Resume Next
    SaveSetting REG_APP, REG_SEC, "Scope", IIf(optSel.Value, "sel", IIf(optAllPages.Value, "all", "page"))
    SaveSetting REG_APP, REG_SEC, "Obj", IIf(optAllObj.Value, "all", "bmp")
    SaveSetting REG_APP, REG_SEC, "How", IIf(optOne.Value, "one", "each")
    SaveSetting REG_APP, REG_SEC, "Mode", IIf(optRGB.Value, "rgb", IIf(optGray.Value, "gray", "cmyk"))
    SaveSetting REG_APP, REG_SEC, "Dpi", cboDpi.Text
    SaveSetting REG_APP, REG_SEC, "AA", IIf(chkAA.Value, "1", "0")
    SaveSetting REG_APP, REG_SEC, "Transp", IIf(chkTransp.Value, "1", "0")
End Sub

Private Sub btnRun_Click()
    Dim pg As Page, sr As ShapeRange

    If Documents.Count = 0 Then
        Status "Hujjat ochilmagan."
        Exit Sub
    End If

    If Not IsNumeric(cboDpi.Text) Then
        Status "Razreshenie raqam bo'lishi kerak (masalan 300)."
        Exit Sub
    End If
    m_Dpi = CLng(cboDpi.Text)
    If m_Dpi < 30 Or m_Dpi > 2400 Then
        Status "Razreshenie 30 dan 2400 gacha bo'lishi kerak."
        Exit Sub
    End If

    If optRGB.Value Then
        m_Mode = cdrRGBColorImage
    ElseIf optGray.Value Then
        m_Mode = cdrGrayscaleImage
    Else
        m_Mode = cdrCMYKColorImage
    End If
    If chkAA.Value Then m_AA = cdrNormalAntiAliasing Else m_AA = cdrNoAntiAliasing
    m_Transp = chkTransp.Value
    m_Bright = scrBright.Value
    m_OnlyBmp = optOnlyBmp.Value
    m_Done = 0: m_Fail = 0: m_Warn = 0

    If optSel.Value Then
        If ActiveSelectionRange.Count = 0 Then
            Status "Hech narsa tanlanmagan."
            Exit Sub
        End If
    End If

    SaveSettings
    Status "Ishlanmoqda..."
    DoEvents

    ActiveDocument.BeginCommandGroup "Manga Bitmap"
    Application.Optimization = True
    Application.EventsEnabled = False
    On Error GoTo Xato

    If optSel.Value Then
        Set sr = ActiveSelectionRange
        ProcessRange sr
    ElseIf optPage.Value Then
        ProcessPage ActivePage
    Else
        For Each pg In ActiveDocument.Pages
            ProcessPage pg
        Next pg
    End If

    GoTo Yakun
Xato:
    m_Fail = m_Fail + 1
    Resume Yakun
Yakun:
    On Error Resume Next
    Application.EventsEnabled = True
    Application.Optimization = False
    ActiveDocument.EndCommandGroup
    ActiveWindow.Refresh
    Application.Refresh

    Dim msg As String
    If m_Done = 0 And m_Fail = 0 Then
        msg = "Bitmap qilinadigan obyekt topilmadi."
    Else
        msg = "Tayyor: " & m_Done & " ta bitmap qilindi."
        If m_Fail > 0 Then msg = msg & "  Xato: " & m_Fail & " ta."
        If m_Warn > 0 Then msg = msg & "  Yorqinlik qo'llanmadi: " & m_Warn & " ta."
    End If
    Status msg
End Sub

Private Sub Status(ByVal txt As String)
    lblStatus.Caption = txt
End Sub

' ===================================================================
'  Asosiy ish
' ===================================================================

' Bitta listni qayta ishlash (faqat ko'rinadigan va tahrirlanadigan qatlamlar).
Private Sub ProcessPage(ByVal pg As Page)
    Dim lr As Layer, s As Shape
    Dim sr As New ShapeRange

    For Each lr In pg.Layers
        If lr.Visible And lr.Editable And Not lr.IsSpecialLayer Then
            For Each s In lr.Shapes
                sr.Add s
            Next s
        End If
    Next lr
    ProcessRange sr
End Sub

' Berilgan yuqori darajadagi obyektlarni tanlangan usul bo'yicha bitmap qilish.
Private Sub ProcessRange(ByVal src As ShapeRange)
    Dim s As Shape
    Dim targets As New ShapeRange

    If optOne.Value Then
        ' Hammasini bitta bitmap: faqat yuqori darajadagi obyektlar guruhlanadi.
        For Each s In src
            If Not s.Locked Then
                If Not m_OnlyBmp Or ContainsBitmap(s) Then targets.Add s
            End If
        Next s
        If targets.Count = 0 Then Exit Sub
        If targets.Count = 1 Then
            ConvertOne targets(1)
        Else
            Dim grp As Shape
            On Error Resume Next
            Set grp = targets.Group
            On Error GoTo 0
            If grp Is Nothing Then
                m_Fail = m_Fail + 1
            Else
                ConvertOne grp
            End If
        End If
    Else
        ' Har birini alohida: avval ro'yxat yig'iladi, keyin o'zgartiriladi.
        ' (bitmap qilinganda obyekt almashadi, shuning uchun alohida ro'yxat)
        Dim list As New Collection, item As Variant
        For Each s In src
            Collect s, targets
        Next s
        For Each s In targets
            list.Add s
        Next s
        For Each item In list
            ConvertOne item
        Next item
    End If
End Sub

' Alohida rejim uchun obyektlarni yig'ish.
' "Faqat rasmlar" bo'lsa guruh va PowerClip ichidagi rasmlar ham topiladi.
Private Sub Collect(ByVal s As Shape, ByVal targets As ShapeRange)
    Dim c As Shape
    If s.Locked Then Exit Sub

    If Not m_OnlyBmp Then
        targets.Add s
        Exit Sub
    End If

    If s.Type = cdrBitmapShape Then
        targets.Add s
    ElseIf s.Type = cdrGroupShape Then
        For Each c In s.Shapes
            Collect c, targets
        Next c
    End If

    If HasPowerClip(s) Then
        For Each c In s.PowerClip.Shapes
            Collect c, targets
        Next c
    End If
End Sub

Private Function ContainsBitmap(ByVal s As Shape) As Boolean
    Dim c As Shape
    If s.Type = cdrBitmapShape Then ContainsBitmap = True: Exit Function
    If s.Type = cdrGroupShape Then
        For Each c In s.Shapes
            If ContainsBitmap(c) Then ContainsBitmap = True: Exit Function
        Next c
    End If
    If HasPowerClip(s) Then
        For Each c In s.PowerClip.Shapes
            If ContainsBitmap(c) Then ContainsBitmap = True: Exit Function
        Next c
    End If
End Function

Private Function HasPowerClip(ByVal s As Shape) As Boolean
    On Error Resume Next
    HasPowerClip = Not (s.PowerClip Is Nothing)
    If Err.Number <> 0 Then HasPowerClip = False
    Err.Clear
End Function

' Bitta obyektni (kerak bo'lsa yorqinlik bilan) bitmap qilish.
Private Sub ConvertOne(ByVal s As Shape)
    Dim target As Shape, g As Shape

    On Error GoTo Xato
    Set target = s
    If m_Bright <> 0 Then
        Set g = AddBrightness(s)
        If Not g Is Nothing Then Set target = g
    End If

    target.ConvertToBitmapEx m_Mode, False, m_Transp, m_Dpi, m_AA, True, False, 95
    m_Done = m_Done + 1
    Exit Sub
Xato:
    m_Fail = m_Fail + 1
End Sub

' Yorqinlik: obyekt ustiga oq (ochroq) yoki qora (to'qroq) shaffof
' to'rtburchak qo'yib, birga guruhlanadi - bitmap qilinganda rangga singadi.
Private Function AddBrightness(ByVal s As Shape) As Shape
    Dim r As Shape, sr As New ShapeRange
    Dim x As Double, y As Double, w As Double, h As Double

    On Error GoTo Xato
    s.GetBoundingBox x, y, w, h
    Set r = s.Layer.CreateRectangle2(x, y, w, h)
    r.Outline.SetNoOutline
    If m_Bright > 0 Then
        r.Fill.UniformColor.RGBAssign 255, 255, 255
    Else
        r.Fill.UniformColor.RGBAssign 0, 0, 0
    End If
    r.Transparency.ApplyUniformTransparency 100 - Abs(m_Bright)
    r.OrderFrontOf s
    sr.Add s
    sr.Add r
    Set AddBrightness = sr.Group
    Exit Function
Xato:
    Resume Tozalash
Tozalash:
    On Error Resume Next
    If Not r Is Nothing Then r.Delete
    m_Warn = m_Warn + 1
    Set AddBrightness = Nothing
End Function
